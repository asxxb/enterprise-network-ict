! ==============================================================================
! SECTION 2: NETWORK ADDRESS TRANSLATION (NAT)
! ==============================================================================

! 1. Dynamic PAT (Port Address Translation) for Corporate Internal LAN
nat (INSIDE_TRANSIT,OUTSIDE) source dynamic OBJ_NET_CORP_CAMPUS interface description Outbound_Corporate_Internet_PAT

! 2. Dynamic PAT for Visitor Guest Wi-Fi (Translates to secondary IP to isolate reputation)
object network OBJ_PUB_IP_GUEST_PAT
 host 198.51.100.3
!
nat (GUEST,OUTSIDE) source dynamic OBJ_NET_GUEST_WIFI OBJ_PUB_IP_GUEST_PAT description Outbound_Guest_Internet_PAT

! 3. Static Destination NAT (Port Forwarding) for Published Corporate Web Portal
nat (OUTSIDE,INSIDE_TRANSIT) source any destination static OBJ_PUB_IP_WEB OBJ_HOST_WEB_PORTAL service https https description Inbound_HTTPS_Web_Portal_DNAT

! 4. Static 1:1 NAT for External SIP Trunking with VoIP ITSP Carrier
nat (INSIDE_TRANSIT,OUTSIDE) source static OBJ_HOST_PBX OBJ_PUB_IP_SIP destination static OBJ_HOST_ITSP_GATEWAY OBJ_HOST_ITSP_GATEWAY description Dedicated_VoIP_SIP_Carrier_NAT

! 5. NAT Exemption (Identity NAT) for Remote Access VPN Clients
nat (INSIDE_TRANSIT,any) source static OBJ_NET_CORP_CAMPUS OBJ_NET_CORP_CAMPUS destination static OBJ_POOL_VPN_CLIENTS OBJ_POOL_VPN_CLIENTS no-proxy-arp route-lookup
!

! SECTION 3: FIREWALL ACCESS CONTROL LISTS (SECURITY POLICIES)

access-list ACL_INSIDE_OUT extended permit tcp object OBJ_NET_CORP_CAMPUS any eq 443
access-list ACL_INSIDE_OUT extended permit udp object OBJ_NET_CORP_CAMPUS any eq 53
access-list ACL_INSIDE_OUT extended permit udp object OBJ_NET_CORP_CAMPUS any eq 123

access-list ACL_INSIDE_OUT extended deny ip object OBJ_NET_ELV_ACCESS any log
access-list ACL_INSIDE_OUT extended deny ip any any log

access-list ACL_GUEST_OUT extended permit tcp object OBJ_NET_GUEST_WIFI any eq 80
access-list ACL_GUEST_OUT extended permit tcp object OBJ_NET_GUEST_WIFI any eq 443

! ==============================================================================
! SECTION 5: CORE SWITCH SVI ACCESS CONTROL LISTS (INTER-VLAN MICRO-SEGMENTATION)
! Applied directly to SVIs on Core-SW-01 and Core-SW-02 for wire-speed filtering
! ==============================================================================

! ------------------------------------------------------------------------------
! 1. CCTV Surveillance Isolation (Applied Inbound on SVI Vlan70)
! ------------------------------------------------------------------------------
ip access-list extended ACL_CCTV_SVI_IN
 ! Permit RTSP streaming and web management to NVR storage interfaces only
 permit tcp 10.10.70.0 0.0.0.255 host 10.10.20.16 eq 554
 permit udp 10.10.70.0 0.0.0.255 host 10.10.20.16 eq 554
 permit tcp 10.10.70.0 0.0.0.255 host 10.10.20.16 eq 80
 permit tcp 10.10.70.0 0.0.0.255 host 10.10.20.16 eq 443
 ! Permit NTP time synchronization with Core Switch VIP
 permit udp 10.10.70.0 0.0.0.255 host 10.10.70.1 eq 123
 ! Deny and log all other outbound traffic (Blocks Internet, workstations, servers)
 deny ip 10.10.70.0 0.0.0.255 any log
!
interface Vlan70
 ip access-group ACL_CCTV_SVI_IN in
!

! ------------------------------------------------------------------------------
! 2. Access Control Isolation (Applied Inbound on SVI Vlan80)
! ------------------------------------------------------------------------------
ip access-list extended ACL_ACCESS_CTRL_SVI_IN
 ! Permit controller communication with central Access Control Server only
 permit tcp 10.10.80.0 0.0.0.255 host 10.10.20.17
 permit udp 10.10.80.0 0.0.0.255 host 10.10.20.17
 ! Permit NTP time synchronization with Core Switch VIP
 permit udp 10.10.80.0 0.0.0.255 host 10.10.80.1 eq 123
 ! Deny all other traffic
 deny ip 10.10.80.0 0.0.0.255 any log
!
interface Vlan80
 ip access-group ACL_ACCESS_CTRL_SVI_IN in
!

! ------------------------------------------------------------------------------
! 3. Printer Isolation (Applied Inbound on SVI Vlan90)
! ------------------------------------------------------------------------------
ip access-list extended ACL_PRINTERS_SVI_IN
 ! Permit SNMP and Status Reporting to Central Print Server
 permit ip 10.10.90.0 0.0.0.255 host 10.10.20.13
 ! Permit NTP time sync
 permit udp 10.10.90.0 0.0.0.255 host 10.10.90.1 eq 123
 ! Deny all other outbound initiated traffic (Printers cannot initiate connections to PCs)
 deny ip 10.10.90.0 0.0.0.255 any log
!
interface Vlan90
 ip access-group ACL_PRINTERS_SVI_IN in
!

! ------------------------------------------------------------------------------
! 4. User Floors Micro-Segmentation (Applied Inbound on SVIs Vlan30, 31, 32, 33)
! ------------------------------------------------------------------------------
ip access-list extended ACL_USERS_SVI_IN
 ! Block standard users from accessing Management Infrastructure (VLAN 10)
 ! Exception: IT Admin Jump Host at 10.10.31.50
 permit ip host 10.10.31.50 10.10.10.0 0.0.0.255
 deny ip any 10.10.10.0 0.0.0.255 log
 ! Block users from accessing raw CCTV cameras and door controllers directly
 deny ip any 10.10.70.0 0.0.0.255 log
 deny ip any 10.10.80.0 0.0.0.255 log
 ! Permit printing ports (RAW 9100 and IPP 631) to Printers
 permit tcp any 10.10.90.0 0.0.0.255 eq 9100
 permit tcp any 10.10.90.0 0.0.0.255 eq 631
 ! Permit standard corporate server access (AD, DNS, DHCP, File, Web)
 permit ip any 10.10.20.0 0.0.0.255
 ! Permit all other outbound Internet traffic via default gateway
 permit ip any any
!
interface Vlan30
 ip access-group ACL_USERS_SVI_IN in
interface Vlan31
 ip access-group ACL_USERS_SVI_IN in
interface Vlan32
 ip access-group ACL_USERS_SVI_IN in
interface Vlan33
 ip access-group ACL_USERS_SVI_IN in
!
